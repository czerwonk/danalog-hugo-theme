{{/* Input: original image. Returns {image, image2x, srcset, placeholder}: image 600px thumbnail with mat image2x 1200px version with a proportionally doubled mat for high-density screens, or "" if the original is too small to provide one (Hugo does not upscale) srcset value for the img srcset attribute placeholder CSS variables for the print-like loading placeholder (see print-placeholder.html) */}} {{ $base := .Filter (slice images.AutoOrient (images.Process "fit 600x600")) }} {{ $large := .Filter (slice images.AutoOrient (images.Process "fit 1200x1200")) }} {{ $image := partial "apply-border.html" (dict "image" $base) }} {{ $image2x := "" }} {{ $srcset := printf "%s 1x" $image.RelPermalink }} {{ if gt $large.Width $base.Width }} {{ $image2x = partial "apply-border.html" (dict "image" $large "scale" 2) }} {{ $srcset = printf "%s, %s 2x" $srcset $image2x.RelPermalink }} {{ end }} {{ $placeholder := partial "print-placeholder.html" (dict "base" $base "image" $image) }} {{ return dict "image" $image "image2x" $image2x "srcset" $srcset "placeholder" $placeholder }}
